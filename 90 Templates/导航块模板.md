<%*
const targetFile = tp.config?.target_file;
const cache = targetFile ? app.metadataCache.getFileCache(targetFile) : null;
const frontmatter = Object.assign({}, tp.frontmatter ?? {}, cache?.frontmatter ?? {});

function valueText(value) {
  if (Array.isArray(value)) {
    return value.filter(Boolean).join(", ");
  }
  if (value === null || value === undefined) {
    return "";
  }
  return String(value).trim();
}

const prev = valueText(frontmatter.prev);
const next = valueText(frontmatter.next);
const lines = [];

if (prev || next) {
  lines.push("## 导航");
  if (prev) {
    lines.push(`- 上一页：${prev}`);
  }
  if (next) {
    lines.push(`- 下一步：${next}`);
  }
}

tR += lines.join("\n");
%>
